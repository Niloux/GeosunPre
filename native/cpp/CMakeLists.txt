cmake_minimum_required(VERSION 3.24)
project(GeosunPreNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GEOSUNPRE_COLMAP_SOURCE "" CACHE PATH "COLMAP 4.2.0 source directory")
if(NOT EXISTS "${GEOSUNPRE_COLMAP_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Run setup-native.ps1 to prepare COLMAP 4.2.0")
endif()
set(GUI_ENABLED OFF CACHE BOOL "" FORCE)
set(MVS_ENABLED OFF CACHE BOOL "" FORCE)
set(CUDA_ENABLED OFF CACHE BOOL "" FORCE)
set(ONNX_ENABLED OFF CACHE BOOL "" FORCE)
set(CGAL_ENABLED OFF CACHE BOOL "" FORCE)
set(DOWNLOAD_ENABLED OFF CACHE BOOL "" FORCE)
set(TESTS_ENABLED OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(FAISS_ENABLE_GPU OFF CACHE BOOL "" FORCE)
add_subdirectory("${GEOSUNPRE_COLMAP_SOURCE}" colmap EXCLUDE_FROM_ALL)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(OpenBLAS CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(OpenCV CONFIG REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(tomlplusplus CONFIG REQUIRED)
add_library(geosunpre_io STATIC runtime.cpp array_io.cpp)
target_link_libraries(geosunpre_io PUBLIC nlohmann_json::nlohmann_json ZLIB::ZLIB)
add_library(geosunpre_scene STATIC scene.cpp)
target_link_libraries(geosunpre_scene PUBLIC geosunpre_io ${OpenCV_LIBS})
add_executable(geosunpre-native main.cpp sfm_pipeline.cpp las_io.cpp registration.cpp joint.cpp delivery.cpp pipeline.cpp)
target_include_directories(geosunpre-native PRIVATE "${GEOSUNPRE_COLMAP_SOURCE}/src" "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(geosunpre-native PRIVATE colmap_exe geosunpre_scene OpenBLAS::OpenBLAS tomlplusplus::tomlplusplus)
if(MSVC)
    target_compile_options(geosunpre-native PRIVATE /W4 /utf-8 /O2)
endif()
set_target_properties(geosunpre-native PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")

set(GEOSUNPRE_TORCH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.venv/Lib/site-packages/torch" CACHE PATH "LibTorch 2.8 CUDA SDK")
if(NOT EXISTS "${GEOSUNPRE_TORCH_ROOT}/include/torch/script.h")
    message(FATAL_ERROR "Prepare the locked Torch 2.8 CUDA libraries, or set GEOSUNPRE_TORCH_ROOT to a LibTorch SDK")
endif()
add_executable(geosunpre-inference inference.cpp models.cpp frontend.cpp)
target_include_directories(geosunpre-inference SYSTEM PRIVATE
    "${GEOSUNPRE_TORCH_ROOT}/include" "${GEOSUNPRE_TORCH_ROOT}/include/torch/csrc/api/include")
target_link_libraries(geosunpre-inference PRIVATE geosunpre_scene CUDA::cudart)
foreach(library torch torch_cpu torch_cuda c10 c10_cuda)
    target_link_libraries(geosunpre-inference PRIVATE "${GEOSUNPRE_TORCH_ROOT}/lib/${library}.lib")
endforeach()
if(MSVC)
    target_compile_options(geosunpre_io PRIVATE /utf-8 /O2)
    target_compile_options(geosunpre-inference PRIVATE /utf-8 /O2 /bigobj)
    target_link_options(geosunpre-inference PRIVATE "/INCLUDE:?warp_size@cuda@at@@YAHXZ")
endif()
set_target_properties(geosunpre-inference PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/inference"
    RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin/inference")
file(GLOB torch_runtime "${GEOSUNPRE_TORCH_ROOT}/lib/*.dll")
list(FILTER torch_runtime EXCLUDE REGEX "torch_python\\.dll$")
foreach(library ${torch_runtime})
    add_custom_command(TARGET geosunpre-inference POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${library}" "$<TARGET_FILE_DIR:geosunpre-inference>")
endforeach()
